08_payload_range
SELECT CASE WHEN PayloadMass < 2000 THEN '<2,000 kg' WHEN PayloadMass < 5000 THEN '2,000–4,999 kg' ELSE '5,000+ kg' END AS payload_band, COUNT(*) AS flights, ROUND(100.0*AVG(Class),1) AS success_pct FROM SPACEXTBL GROUP BY payload_band ORDER BY flights DESC
  payload_band  flights  success_pct
2,000–4,999 kg       29         55.2
     5,000+ kg       14         35.7
     <2,000 kg       13         23.1
